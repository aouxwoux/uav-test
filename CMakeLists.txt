cmake_minimum_required(VERSION 3.16)
project(ideaforge_uav_localizer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs features2d calib3d)
add_library(uavloc src/localizer.cpp src/io.cpp)
target_include_directories(uavloc PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(uavloc PUBLIC ${OpenCV_LIBS})
if(MSVC)
  target_compile_options(uavloc PRIVATE /W4)
else()
  target_compile_options(uavloc PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(uav_localizer src/main.cpp)
target_link_libraries(uav_localizer PRIVATE uavloc)
add_library(demo_data tools/demo_data.cpp)
target_link_libraries(demo_data PUBLIC uavloc)
add_executable(make_demo tools/make_demo.cpp)
target_link_libraries(make_demo PRIVATE demo_data)
add_executable(run_demo tools/run_demo.cpp)
target_link_libraries(run_demo PRIVATE demo_data)
include(CTest)
if(BUILD_TESTING)
  add_executable(localizer_tests tests/test_localizer.cpp)
  target_link_libraries(localizer_tests PRIVATE uavloc)
  add_test(NAME localizer_tests COMMAND localizer_tests)
  add_test(NAME cli_demo_generate COMMAND make_demo ${CMAKE_BINARY_DIR}/cli-demo)
  set_tests_properties(cli_demo_generate PROPERTIES FIXTURES_SETUP demo_data)
  add_test(NAME cli_batch COMMAND uav_localizer ${CMAKE_BINARY_DIR}/cli-demo/reference ${CMAKE_BINARY_DIR}/cli-demo/input ${CMAKE_BINARY_DIR}/cli-demo/results.csv)
  set_tests_properties(cli_batch PROPERTIES FIXTURES_REQUIRED demo_data)
  add_test(NAME cli_demo_cleanup COMMAND ${CMAKE_COMMAND} -E rm -rf ${CMAKE_BINARY_DIR}/cli-demo)
  set_tests_properties(cli_demo_cleanup PROPERTIES FIXTURES_CLEANUP demo_data)
  add_test(NAME visual_demo COMMAND ${CMAKE_COMMAND}
    -DDEMO_EXE=$<TARGET_FILE:run_demo> -DOUTPUT=${CMAKE_BINARY_DIR}/visual-demo
    -P ${CMAKE_SOURCE_DIR}/tests/visual_demo.cmake)
endif()
